cmake_minimum_required(VERSION 3.27)

# This directory is a standalone experiment. No product target includes or
# links it, even when the root's documentation-only option is enabled.
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Configure tools/spatial_gpu as a standalone source directory")
endif()
project(bike_spatial_gpu_tools LANGUAGES CXX)
enable_testing()

get_filename_component(_gpu_repo_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(READ "${_gpu_repo_root}/native/dependencies.lock.json" _gpu_dependency_lock)
string(JSON _gpu_rapidjson_directory GET "${_gpu_dependency_lock}"
  dependencies rapidjson extracted_root)
set(SPATIAL_GPU_RAPIDJSON_ROOT "${_gpu_repo_root}/.deps/${_gpu_rapidjson_directory}"
  CACHE PATH "Already acquired, locked RapidJSON tree for the existing bt harness")

# Reuse the existing bt implementation exactly. Its cold manifest support
# references JSON/SHA-256, so compile those stdlib/header-only sources locally
# rather than linking a product target or creating a replacement test harness.
# Only the required, already acquired RapidJSON headers enter this graph.
foreach(_gpu_header IN ITEMS "include/rapidjson/document.h" "include/rapidjson/reader.h")
  if(NOT EXISTS "${SPATIAL_GPU_RAPIDJSON_ROOT}/${_gpu_header}")
    message(FATAL_ERROR "Missing locked RapidJSON header: ${SPATIAL_GPU_RAPIDJSON_ROOT}/${_gpu_header}")
  endif()
  string(JSON _gpu_expected_hash GET "${_gpu_dependency_lock}"
    dependencies rapidjson required_files "${_gpu_header}")
  file(SHA256 "${SPATIAL_GPU_RAPIDJSON_ROOT}/${_gpu_header}" _gpu_actual_hash)
  if(NOT _gpu_actual_hash STREQUAL _gpu_expected_hash)
    message(FATAL_ERROR "Locked RapidJSON header digest mismatch: ${_gpu_header}")
  endif()
endforeach()

# migration_test.cpp uses this generated header for its unused fixture helpers.
# No fixture is read by decision_test; configure/build perform no simulations.
foreach(_gpu_path IN ITEMS "${_gpu_repo_root}")
  string(FIND "${_gpu_path}" "\"" _gpu_quote)
  string(FIND "${_gpu_path}" "\\" _gpu_backslash)
  string(FIND "${_gpu_path}" "\n" _gpu_newline)
  string(FIND "${_gpu_path}" "\r" _gpu_carriage_return)
  if(NOT _gpu_quote EQUAL -1 OR NOT _gpu_backslash EQUAL -1
      OR NOT _gpu_newline EQUAL -1 OR NOT _gpu_carriage_return EQUAL -1)
    message(FATAL_ERROR "Source path cannot be embedded as a C++ string")
  endif()
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/support/bike_migration_paths.hpp"
  CONTENT "// Generated by tools/spatial_gpu; do not edit.\n#pragma once\n#define BIKE_FIXTURE_ROOT \"${_gpu_repo_root}/tests/fixtures/migration\"\n#define BIKE_ORACLE_MANIFEST \"${_gpu_repo_root}/migration/oracle-manifest.json\"\n")

add_executable(spatial_gpu_decision_test
  decision_test.cpp
  ../../native/tests/support/migration_test.cpp
  ../../native/src/io/json.cpp
  ../../native/src/io/hash.cpp)
set_target_properties(spatial_gpu_decision_test PROPERTIES
  CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(spatial_gpu_decision_test PRIVATE
  "${_gpu_repo_root}/native/src" "${_gpu_repo_root}/native/tests"
  "${CMAKE_CURRENT_BINARY_DIR}/support")
target_include_directories(spatial_gpu_decision_test SYSTEM PRIVATE
  "${SPATIAL_GPU_RAPIDJSON_ROOT}/include")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(spatial_gpu_decision_test PRIVATE -ffp-contract=off -Wall -Wextra)
endif()
add_test(NAME spatial.gpu_decision COMMAND spatial_gpu_decision_test)
set_tests_properties(spatial.gpu_decision PROPERTIES LABELS "spatial;gpu;unit")
