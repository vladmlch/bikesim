## Global Constraints

- Только mid-drive. Bosch CX Gen 4 по памяти: 85 Н·м; 250 Вт номинал, ~600 Вт пик; лаг момента 40 мс; Eco 0,6 / Tour 1,4 / eMTB 1,4–3,4 / Turbo 3,4; по умолчанию Turbo; каденс до 120 об/мин; спад перед 25 км/ч, ноль на отсечке.
- Gate: ассист только при вращении шатуна вперёд > 5°/с и положительном измеренном моменте стопы; `control.motor_torque_nm` — потолок, не газ.
- Датчики Gen 4 без ABS: момент (1 кГц), каденс, заднее колесо 1 имп./оборот, скорость вала, ток; IMU и переднее колесо недоступны и помечены маской. Истина не утекает.

- Одна односторонняя муфта (freehub) и обгонная муфта мотора (ротор → звезда); отрицательного ведущего момента на колесо нет. См. план 2026-10-04-realistic-pedelec-drive.md.

```python
"""Mid-drive profiles as declared data. Numbers recalled from memory on 2026-10-03
at the user's instruction (no web lookup); they are unverified and say so."""
from dataclasses import dataclass
import math


@dataclass(frozen=True)
class MotorProfile:
    name: str
    peak_torque_nm: float
    rated_power_w: float
    peak_power_w: float
    torque_tau_s: float
    mode_gains: dict
    emtb_full_gain_at_nm: float
    cadence_support_max_rpm: float
    cutoff_mps: float
    taper_width_mps: float
    gate_min_crank_rad_s: float
    provenance: str

    def __post_init__(self):
        values = (self.peak_torque_nm, self.rated_power_w, self.peak_power_w, self.torque_tau_s,
                  self.emtb_full_gain_at_nm, self.cadence_support_max_rpm, self.cutoff_mps,
                  self.taper_width_mps, self.gate_min_crank_rad_s)
        if not all(math.isfinite(v) and v > 0 for v in values):
            raise ValueError('motor profile numbers must be positive and finite')
        if set(self.mode_gains) != {'eco', 'tour', 'emtb', 'turbo'}:
            raise ValueError('profile must declare eco, tour, emtb and turbo')
        if not self.provenance.strip():
            raise ValueError('motor profile requires provenance')


BOSCH_CX_GEN4 = MotorProfile(
    name='bosch_performance_line_cx_gen4',
    peak_torque_nm=85., rated_power_w=250., peak_power_w=600., torque_tau_s=.04,
    mode_gains={'eco': .6, 'tour': 1.4, 'emtb': (1.4, 3.4), 'turbo': 3.4},
    emtb_full_gain_at_nm=40., cadence_support_max_rpm=120.,
    cutoff_mps=25/3.6, taper_width_mps=2/3.6, gate_min_crank_rad_s=math.radians(5.),
    provenance='Bosch Performance Line CX Gen 4 (no ABS); recalled from memory 2026-10-03, '
               'unverified, no web lookup by user instruction')


def assist_gain(profile: MotorProfile, mode: str, human_nm: float) -> float:
    if mode not in profile.mode_gains:
        raise ValueError(f'unknown assist mode {mode!r}')
    gain = profile.mode_gains[mode]
    if isinstance(gain, tuple):
        low, high = gain
        # eMTB approximation: multiplier rises linearly with rider torque to the
        # declared saturation torque. The real law is proprietary.
        fraction = min(1., max(0., human_nm)/profile.emtb_full_gain_at_nm)
        return low + (high-low)*fraction
    return float(gain)
```

```python
# motor.py — gate as a hard permission after the torque dynamics
def pedelec_cap(human_nm, crank_rad_s, speed_mps, gain, external_cap_nm, *, braking,
                taper_start_mps, cutoff_mps, gate_min_crank_rad_s):
    if not all(isfinite(x) for x in (human_nm, crank_rad_s, speed_mps, gain, taper_start_mps,
                                    cutoff_mps, gate_min_crank_rad_s)):
        raise ValueError('nonfinite pedelec state')
    if gain < 0 or not 0 <= taper_start_mps < cutoff_mps or gate_min_crank_rad_s < 0:
        raise ValueError('invalid pedelec envelope')
    if external_cap_nm is not None and (not isfinite(external_cap_nm) or external_cap_nm < 0):
        raise ValueError('invalid external motor ceiling')
    if braking or crank_rad_s <= gate_min_crank_rad_s or human_nm <= 0:
        return 0.
    taper = min(1., max(0., (cutoff_mps-abs(speed_mps))/(cutoff_mps-taper_start_mps)))
    cap = gain*human_nm*taper
    return cap if external_cap_nm is None else min(cap, external_cap_nm)
```

```python
# sim/research/sensor_profile.py
from dataclasses import dataclass


@dataclass(frozen=True)
class SensorMask:
    specific_force: bool
    pitch_rate: bool
    front_wheel: bool
    rear_wheel: bool
    crank: bool
    motor_torque: bool
    human_torque: bool
    motor_shaft: bool
    motor_current: bool


PROFILES = {
    'bosch_cx_gen4_no_abs': SensorMask(False, False, False, True, True, True, True, True, True),
    'generic_imu': SensorMask(True, True, True, True, True, True, True, True, True),
}


class PulseWheelSpeed:
    def __init__(self, pulses_per_rev: int, radius_m: float):
        if type(pulses_per_rev) is not int or pulses_per_rev < 1 or radius_m <= 0:
            raise ValueError('pulse sensor needs >=1 pulse per revolution and a positive radius')
        self.pulses_per_rev = pulses_per_rev
        self.radius_m = float(radius_m)
        self._last_pulse_time = None
        self._last_pulse_index = None
        self._speed = None

    def update(self, angle_rad, time_s):
        import math
        index = math.floor(angle_rad*self.pulses_per_rev/(2*math.pi))
        if self._last_pulse_index is None:
            self._last_pulse_index, self._last_pulse_time = index, time_s
            return None
        if index > self._last_pulse_index:
            turns = (index-self._last_pulse_index)/self.pulses_per_rev
            self._speed = 2*math.pi*self.radius_m*turns/(time_s-self._last_pulse_time)
            self._last_pulse_index, self._last_pulse_time = index, time_s
        return self._speed
```

```python
from dataclasses import dataclass, field
from math import isfinite


@dataclass
class ShiftInterlock:
    teeth: int
    shift_time_s: float = .3
    unload_fraction_easier: float = .8
    unload_fraction_harder: float = .3
    recover_s: float = .2
    phase: str = 'idle'
    requested: int | None = None
    reference_load_nm: float = 0.
    elapsed_s: float = 0.

    def __post_init__(self):
        if type(self.teeth) is not int or self.teeth <= 0:
            raise ValueError('invalid cassette teeth')
        for name in ('shift_time_s', 'recover_s'):
            if not isfinite(getattr(self, name)) or getattr(self, name) <= 0:
                raise ValueError(f'{name} must be positive')
        for name in ('unload_fraction_easier', 'unload_fraction_harder'):
            if not 0 <= getattr(self, name) <= 1:
                raise ValueError(f'{name} lies in [0, 1]')

    @property
    def direction(self):
        if self.requested is None:
            return None
        return 'easier' if self.requested > self.teeth else 'harder'

    @property
    def rider_factor(self) -> float:
        if self.phase == 'unload':
            return (self.unload_fraction_easier if self.direction == 'easier'
                    else self.unload_fraction_harder)
        if self.phase == 'recover':
            start = self.unload_fraction_easier if self._last_direction == 'easier' else self.unload_fraction_harder
            return min(1., start + (1.-start)*self.elapsed_s/self.recover_s)
        return 1.

    @property
    def motor_allowed(self) -> bool:
        # Gen 4 with a mechanical derailleur has no shift detection: the motor
        # keeps following rider torque through its own lag.
        return True

    def request(self, teeth: int, load_nm: float) -> None:
        if type(teeth) is not int or teeth <= 0 or not isfinite(load_nm):
            raise ValueError('invalid shift request')
        if self.phase != 'idle' or teeth == self.teeth:
            return
        self.requested, self.reference_load_nm = teeth, abs(load_nm)
        self.phase, self.elapsed_s = 'unload', 0.
        self._last_direction = self.direction

    def observe(self, load_nm: float, dt_s: float) -> int | None:
        if not (isfinite(load_nm) and isfinite(dt_s)) or dt_s <= 0:
            raise ValueError('invalid solved shift state')
        self.elapsed_s += dt_s
        if self.phase == 'recover' and self.elapsed_s >= self.recover_s:
            self.phase, self.elapsed_s = 'idle', 0.
            return None
        if self.phase != 'unload' or self.elapsed_s < self.shift_time_s:
            return None
        allowed = self.rider_factor*self.reference_load_nm
        if abs(load_nm) > allowed + 1e-9:
            return None
        if self.requested is None:
            raise RuntimeError('shift commit without a pending ratio')
        self.teeth, self.requested = self.requested, None
        self.phase, self.elapsed_s = 'recover', 0.
        return self.teeth
```

`_last_direction` инициализировать в `__post_init__` как `None`. Сравнение `elapsed_s < shift_time_s` при накоплении по 0,01 с даёт `0.30` с ошибкой округления; использовать `self.elapsed_s < self.shift_time_s - 1e-12`.

```python
"""Rider-side gear choice from the road preview: shift before the load rises."""
import math


def _cadence_rpm(speed_mps, wheel_radius_m, front, rear):
    return speed_mps/wheel_radius_m*rear/front*60./(2*math.pi)


def _crank_torque_for_grade(mass_kg, grade, wheel_radius_m, front, rear):
    # Static climbing torque at the crank, muscles only, no motor: the rider
    # chooses a gear they could turn alone if the assist faded.
    return mass_kg*9.81*math.sin(math.atan(grade))*wheel_radius_m*front/rear


def select_gear(cassette, front_teeth, current_teeth, *, speed_mps, wheel_radius_m,
                grade_now, grade_preview, cadence_band_rpm, mass_kg, available_crank_nm):
    if current_teeth not in cassette:
        raise ValueError('current cog is not on the cassette')
    low, high = cadence_band_rpm
    if not 0 < low < high:
        raise ValueError('invalid cadence band')
    grade = max(grade_now, grade_preview)
    candidates = []
    for rear in cassette:
        cadence = _cadence_rpm(max(speed_mps, .5), wheel_radius_m, front_teeth, rear)
        torque = _crank_torque_for_grade(mass_kg, grade, wheel_radius_m, front_teeth, rear)
        in_band = low <= cadence <= high
        feasible = torque <= available_crank_nm
        candidates.append((feasible, in_band, -abs(cadence-(low+high)/2), rear))
    best = max(candidates)
    return best[3]
```
