# src/bike_sim/physics/reference_statics.py
from math import atan, cos, sin, isfinite
import numpy as np


def static_axle_normals(mass_kg, wheelbase_m, com_forward_m, com_height_m, grade):
    values = (mass_kg, wheelbase_m, com_forward_m, com_height_m, grade)
    if not all(isfinite(v) for v in values) or min(mass_kg, wheelbase_m) <= 0:
        raise ValueError('invalid static geometry')
    angle = atan(grade)
    weight = mass_kg*9.81
    front = weight*(com_forward_m*cos(angle)-com_height_m*sin(angle))/wheelbase_m
    return front, weight*cos(angle)-front


def compiled_bike_mass_kg(model) -> float:
    total = 0.
    for i in range(model.nbody):
        name = model.body(i).name
        if i == 0 or name.startswith('rider_'):
            continue
        total += float(model.body_mass[i])
    return total
```

# src/bike_sim/validation/reference_outcomes.py
"""Causal outcome classification. Thresholds here serve the report only; none of
them reaches the rider program, the motor or the external policy."""
from dataclasses import dataclass
from math import isfinite, radians

WHEELIE_MIN_S = .3
WHEELIE_MIN_PITCH_RAD = radians(15.)
TERRAIN_STEP_M = .02
MOTOR_FRACTION = .5
PEDAL_NM = 40.


@dataclass(frozen=True)
class LiftEvent:
    onset_s: float
    end_s: float
    max_relative_pitch_rad: float
    max_clearance_m: float
    cause: str
    motor_nm_at_onset: float
    human_nm_at_onset: float
    road_step_m_at_onset: float

    def is_wheelie(self) -> bool:
        return (self.end_s-self.onset_s >= WHEELIE_MIN_S
                or self.max_relative_pitch_rad >= WHEELIE_MIN_PITCH_RAD)


@dataclass(frozen=True)
class OutcomeEvent:
    kind: str
    onset_s: float
    confirmed_s: float

    def __post_init__(self):
        allowed = {'stall', 'traction_loss', 'wheelie', 'rollback', 'crash', 'flight', 'loop_out'}
        if self.kind not in allowed or not all(isfinite(t) for t in (self.onset_s, self.confirmed_s)):
            raise ValueError('invalid outcome event')
        if not 0 <= self.onset_s <= self.confirmed_s:
            raise ValueError('invalid event chronology')


def attribute_lift_cause(*, road_step_m, motor_nm, motor_peak_nm, human_nm):
    terrain = road_step_m >= TERRAIN_STEP_M
    motor = motor_nm >= MOTOR_FRACTION*motor_peak_nm
    if terrain and motor:
        return 'mixed'
    if terrain:
        return 'terrain'
    if motor:
        return 'motor'
    # Neither a road step nor a motor surge: the remaining source is the legs,
    # whether the pedal torque was large or not. PEDAL_NM is reported, not decisive.
    return 'pedals'


def loop_out(*, com_x_m, rear_contact_x_m, pitch_rate_up_rad_s, front_clearance_m):
    return (front_clearance_m > 0. and com_x_m < rear_contact_x_m
            and pitch_rate_up_rad_s > 0.)


def choose_outcome(events, goal_s, invalid):
    if invalid is not None:
        return 'invalid_numerics' if invalid.startswith('numerics:') else 'invalid_controller'
    terminal = {'crash', 'loop_out'}
    eligible = [e for e in events if goal_s is None or e.confirmed_s <= goal_s]
    if not eligible:
        return 'completed' if goal_s is not None else 'incomplete'
    first = min(eligible, key=lambda e: (e.confirmed_s, e.onset_s, e.kind))
    return 'crash' if first.kind in terminal else first.kind
```

`attribute_lift_cause` для `human_nm < PEDAL_NM` без террейна и мотора тоже возвращает `'pedals'` — отрыв без видимой причины относится к педалям как к единственному оставшемуся источнику; это диагностика. `WheelieTracker` закрывает эпизод `front_lift` и возвращает `LiftEvent` с данными на onset (момент мотора, момент стоп, разность высот дороги под передним колесом за последние 0,1 с). `PhysicalRuntime` проверяет `loop_out(...)` каждый период по `raw.contact_truth` (COM и заднее пятно уже есть в `WheelieTruth`) и `physical_contact_crash` для касания не колесом; оба — терминал `crash`/`loop_out`. `ResearchEnvironment`: `wheelie` больше не ставит `done`; события пишутся в `episode` JSON (`lift_events`), исход — `choose_outcome` при завершении. Пороги не читаются ни `rider_reaction` (у неё свои, стратегические), ни политикой.

LEGAL_OUTCOMES = {'completed', 'stall', 'traction_loss', 'wheelie', 'rollback', 'flight'}


def reference_ready(case: dict) -> bool:
    required = ('budgets_ok', 'energy_ok', 'refinement_ok', 'realtime_ok')
    return case.get('outcome') in LEGAL_OUTCOMES and all(case.get(k) is True for k in required)


def release_level(case: dict) -> str:
    return 'biomechanically_validated' if case.get('evidence_verified') is True else 'mechanically_checked'
