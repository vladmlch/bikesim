# Contract-test harness and domain regression executables. One executable
# and one CTest name per boundary: <domain>_<topic> <- <domain>/test_<topic>.cpp.
# Registered only in the repository-root build; targets get
# warning/hardening/sanitizer policy from bike_apply_native_policy directly.

# The fixture root is computed from this directory up to the repository root
# (native/tests/../..) — never from the process working directory or a
# nested PROJECT_SOURCE_DIR.
get_filename_component(_test_repo_root
  "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(_test_fixture_root "${_test_repo_root}/tests/fixtures")
if(NOT IS_DIRECTORY "${_test_fixture_root}")
  message(FATAL_ERROR
    "fixture tree is missing: ${_test_fixture_root}")
endif()
# The root is embedded into the generated bike_fixture_paths.hpp as a C
# string literal; quotes or backslashes would break that literal.
if(_test_fixture_root MATCHES "[\"'\\\\]")
  message(FATAL_ERROR
    "fixture root must not contain quotes or backslashes: ${_test_fixture_root}")
endif()

# Standalone harness: stdlib-only plus the owned bike::io boundary (strict
# JSON + SHA-256 for fixture digests) — no MuJoCo includes. The support TU
# owns BIKE_FIXTURE_ROOT privately so test targets and the standalone-header
# sweep never see it.
add_library(bike_test_fixtures STATIC
  support/fixture_test.cpp
  support/fixture_test.hpp)
set_target_properties(bike_test_fixtures PROPERTIES
  CXX_STANDARD 23
  CXX_STANDARD_REQUIRED ON
  CXX_EXTENSIONS OFF)
# The fixture root reaches the support TU as a generated
# header, not quoted compile definitions: the check-context writer embeds target
# properties verbatim, and a quote inside one would make native_check_context.json
# invalid. The header is private to bike_test_fixtures.
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/support/bike_fixture_paths.hpp"
  CONTENT "// Generated by native/tests/CMakeLists.txt; do not edit.\n#pragma once\n#define BIKE_FIXTURE_ROOT \"${_test_fixture_root}\"\n")
target_include_directories(bike_test_fixtures PRIVATE
  "${CMAKE_CURRENT_BINARY_DIR}/support")
target_include_directories(bike_test_fixtures PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(bike_test_fixtures SYSTEM PRIVATE
  "${_test_repo_root}/native/src")
target_link_libraries(bike_test_fixtures PUBLIC bike_io)
bike_apply_native_policy(bike_test_fixtures)

# allocation_faults.cpp is the throwing-allocation override TU: it defines
# replacement operator new/delete plus the counting seam, so it must land
# in test binaries directly, never in a shipped library. Two consumers
# share it — native_contract_tests and config_writer —
# and the ODR sweep compiles every compile-database entry once before a
# combined relocatable link, so per-target duplicate compiles would
# collide. One OBJECT library = one compile entry; each executable gets
# the same object through $<TARGET_OBJECTS:...> exactly like a listed
# source.
add_library(bike_test_allocation_faults OBJECT allocation_faults.cpp)
set_target_properties(bike_test_allocation_faults PROPERTIES
  CXX_STANDARD 23
  CXX_STANDARD_REQUIRED ON
  CXX_EXTENSIONS OFF)
  bike_apply_native_policy(bike_test_allocation_faults)

# One subdirectory per contract domain; each registers its own executables
# and CTest names.
foreach(_domain IN ITEMS batch cli config drivetrain engine initialization
    io kinematics legacy model numerics platform random recording replay
    research rider runtime terrain validation viewer viz)
  add_subdirectory(${_domain})
endforeach()
